cmake_minimum_required(VERSION 3.20)
project(backblast VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(backblast_core src/game.cpp)
target_include_directories(backblast_core PUBLIC src)
target_compile_options(backblast_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(backblast src/main.cpp src/terminal.cpp)
target_link_libraries(backblast PRIVATE backblast_core)
target_compile_options(backblast PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS backblast RUNTIME DESTINATION bin)
enable_testing()
add_executable(backblast_tests tests/game_test.cpp src/terminal.cpp)
target_link_libraries(backblast_tests PRIVATE backblast_core)
add_test(NAME game_physics_and_endless_world COMMAND backblast_tests)
