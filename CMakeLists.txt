cmake_minimum_required(VERSION 3.20)
project(zapy VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(zapy_core src/game.cpp)
target_include_directories(zapy_core PUBLIC src)
target_compile_options(zapy_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(zapy src/main.cpp src/terminal.cpp)
target_link_libraries(zapy PRIVATE zapy_core)
target_compile_options(zapy PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS zapy RUNTIME DESTINATION bin)
enable_testing()
add_executable(zapy_tests tests/game_test.cpp src/terminal.cpp)
target_link_libraries(zapy_tests PRIVATE zapy_core)
add_test(NAME game_physics_and_endless_world COMMAND zapy_tests)
add_executable(zapy_render_tests tests/render_test.cpp src/terminal.cpp)
target_link_libraries(zapy_render_tests PRIVATE zapy_core)
target_compile_options(zapy_render_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME rendering_regressions COMMAND zapy_render_tests)
